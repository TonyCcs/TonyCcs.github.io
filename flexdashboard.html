data("ny_noaa")

ny_noaa_df =
  ny_noaa |>
  mutate(
    year = lubridate::year(date),
    month = lubridate::month(date),
    day = lubridate::day(date),
    prcp = prcp / 10,
    tmax = as.numeric(tmax) / 10,
    tmin = as.numeric(tmin) / 10
  )

Scatterplot showing average max temperature in January and in July across all stations

ny_noaa_df |>
  filter(month %in% c(1, 7)) |>
  mutate(month = month.name[month]) |>
  group_by(year, month) |>
  summarize(mean_tmax = mean(tmax, na.rm = TRUE)) |>
  ungroup() |>
  plot_ly(
    x = ~year, y = ~mean_tmax, type = "scatter", 
    mode = "markers", color = ~month
  ) |>
  layout(
    xaxis = list(title = "Year"),
    yaxis = list(title = "Average max temperature (C)")
  )

Boxplot showing average max temperature from each station by month in 2010

ny_noaa_df |>
  filter(year == 2010) |>
  drop_na(tmax) |>
  mutate(month = factor(month.name[month], levels = month.name)) |>
  group_by(id, month) |>
  summarize(mean_tmax = mean(tmax)) |>
  ungroup() |>
  plot_ly(
    x = ~month, y = ~mean_tmax, type = "box", 
    color = ~month, colors = "viridis"
  ) |>
  layout(
    xaxis = list(title = "Month"),
    yaxis = list(title = "Average max temperature (C)"),
    showlegend = FALSE
  )
